{{- /* Renders the contact form. The endpoint is resolved from (in order): - the page's own `formAction` front matter - site params: contactFormAction, formspree_url, formspreeURL The form is always rendered. If no endpoint is configured, it is rendered without an action attribute and a build-time warning is emitted so the mistake is visible in the build log instead of hiding the form. */ -}} {{- $action := trim (string (or .Params.formAction .Site.Params.contactFormAction .Site.Params.formspree_url .Site.Params.formspreeURL)) " " -}} {{- $placeholders := slice "" "your formspree url here" "https://formspree.io/f/your-form-id" -}} {{- if in $placeholders $action -}} {{- warnf "Contact form endpoint is not configured. Set params.contactFormAction to your Formspree URL (page: %s)." .RelPermalink -}} {{- $action = "" -}} {{- end -}}
